def blocked(observer,blocker,relation,mode,granted):
 deadline=time.monotonic()+8
 while time.monotonic()<deadline:
  if observer.execute('select exists(select 1 from pg_stat_activity a join pg_locks l using(pid) where %s=any(pg_blocking_pids(a.pid)) and l.relation=to_regclass(%s) and l.mode=%s and l.granted=%s)',(blocker,relation,mode,granted)).fetchone()[0]:return True
  time.sleep(.01)
 return False
